
=================================================
REDESIGNED NETWORK - ROUTER R1
=================================================

enable
configure terminal
hostname R1

! LABS LAN
interface gigabitEthernet 0/0
description LABS_NETWORK
ip address 192.168.10.1 255.255.255.128
no shutdown
exit

! HOSTEL LAN
interface gigabitEthernet 0/1
description HOSTEL_NETWORK
ip address 192.168.11.1 255.255.255.192
no shutdown
exit

end
write memory

! VERIFICATION
show ip interface brief
show ip route
show running-config
