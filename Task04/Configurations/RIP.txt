
! =====================================
! ROUTER A
! =====================================
enable
configure terminal
hostname A

interface gigabitEthernet 0/0
ip address 10.0.12.1 255.255.255.252
no shutdown
exit

interface gigabitEthernet 0/1
ip address 10.0.13.1 255.255.255.252
no shutdown
exit

router rip
version 2
no auto-summary
network 10.0.0.0
end
write memory
