
! =====================================
! ROUTER R1
! =====================================
enable
configure terminal
hostname R1

interface gigabitEthernet 0/0
ip address 10.0.12.1 255.255.255.252
ip ospf 1 area 0
ip ospf cost 10
no shutdown
exit

interface gigabitEthernet 0/1
ip address 10.0.13.1 255.255.255.252
ip ospf 1 area 0
ip ospf cost 5
no shutdown
exit

router ospf 1
auto-cost reference-bandwidth 10000
end
write memory
